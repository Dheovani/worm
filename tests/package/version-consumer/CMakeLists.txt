cmake_minimum_required(VERSION 3.20)
project(WormVersionConsumer LANGUAGES NONE)

find_package(Worm ${REQUEST} CONFIG QUIET PATHS "${PACKAGE_DIR}" NO_DEFAULT_PATH)
if(EXPECT_FOUND)
  if(NOT Worm_FOUND OR NOT Worm_VERSION STREQUAL EXPECT_VERSION)
    message(FATAL_ERROR "Expected Worm ${EXPECT_VERSION} for '${REQUEST}', got '${Worm_VERSION}' (found=${Worm_FOUND}).")
  endif()
  if(NOT Worm_CONSIDERED_VERSIONS STREQUAL EXPECT_VERSION)
    message(FATAL_ERROR "Version file reported '${Worm_CONSIDERED_VERSIONS}', expected '${EXPECT_VERSION}'.")
  endif()
elseif(Worm_FOUND)
  message(FATAL_ERROR "Unexpectedly accepted '${REQUEST}' from ${PACKAGE_DIR}.")
endif()
